SOC301 3 Credit Hrs

Classical Sociological Theories

Instructor: Dr. B. S. Rayamajhi • B.A. 3rd Year (Sem VI)

In-depth study of Karl Marx, Émile Durkheim, Max Weber, and Georg Simmel. Focuses on structural functionalism, conflict dynamics, and historical contextualization of Terai social structures.

HIST201 3 Credit Hrs

Buddhist Archaeology & Ancient Kapilvastu

Instructor: Prof. H. R. Sharma • B.A. 2nd Year (Sem IV)

Archaeological records of Tilaurakot-Kapilvastu, Kudan (Nigrodharama), Gotihawa, Niglihawa, and Sagarhawa. Stratigraphic interpretations, Buddhist textual accounts, and Ashokan pillar epigraphy.

Heritage Lab Fieldwork
RD302 3 Credit Hrs

Rural Economy & Sustainable Development

Instructor: Lecturer D. K. Yadav • B.A. 3rd Year (Sem VI)

Rural socio-economic structures in Nepal, agrarian transitions, micro-credit institutions, wetland ecosystem livelihoods at Jagdishpur Reservoir, and sustainable rural community development.

    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 2. USERS
CREATE TABLE users (
    user_id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    email VARCHAR(255) UNIQUE NOT NULL,
    password_hash VARCHAR(255) NOT NULL,
    full_name VARCHAR(150) NOT NULL,
    full_name_ne VARCHAR(150),
    role_id INT NOT NULL REFERENCES roles(role_id) ON DELETE RESTRICT,
    department VARCHAR(100) DEFAULT 'HSSD',
    is_active BOOLEAN DEFAULT TRUE,
    last_login TIMESTAMP WITH TIME ZONE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 3. ACADEMIC YEARS & SEMESTERS
CREATE TABLE academic_sessions (
    session_id SERIAL PRIMARY KEY,
    session_name_bs VARCHAR(20) NOT NULL, -- e.g. '2081/2082'
    session_name_ad VARCHAR(20) NOT NULL, -- e.g. '2026/2027'
    is_current BOOLEAN DEFAULT FALSE,
    start_date_bs VARCHAR(15),
    end_date_bs VARCHAR(15)
);

-- 4. ACADEMIC PROGRAMS
CREATE TABLE programs (
    program_id SERIAL PRIMARY KEY,
    program_code VARCHAR(20) UNIQUE NOT NULL, -- e.g. 'BA', 'MA_SOC', 'MA_RD'
    program_name VARCHAR(150) NOT NULL,
    level VARCHAR(50) NOT NULL, -- 'Undergraduate', 'Graduate'
    duration_years INT DEFAULT 4,
    system_type VARCHAR(20) DEFAULT 'Semester' -- 'Semester', 'Annual'
);

-- 5. COURSES
CREATE TABLE courses (
    course_id SERIAL PRIMARY KEY,
    course_code VARCHAR(20) UNIQUE NOT NULL, -- e.g. 'SOC301', 'HIST201'
    course_title VARCHAR(200) NOT NULL,
    program_id INT REFERENCES programs(program_id),
    credit_hours INT NOT NULL,
    semester_level INT NOT NULL,
    syllabus_summary TEXT,
    is_active BOOLEAN DEFAULT TRUE
);

-- 6. FACULTY ALLOCATIONS
CREATE TABLE faculty_allocations (
    allocation_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(course_id) ON DELETE CASCADE,
    faculty_user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    session_id INT REFERENCES academic_sessions(session_id),
    weekly_hours INT DEFAULT 3,
    room_number VARCHAR(50)
);

-- 7. STUDENT ENROLLMENTS
CREATE TABLE enrollments (
    enrollment_id SERIAL PRIMARY KEY,
    student_user_id UUID REFERENCES users(user_id) ON DELETE CASCADE,
    course_id INT REFERENCES courses(course_id),
    session_id INT REFERENCES academic_sessions(session_id),
    roll_number VARCHAR(50) NOT NULL,
    registration_number VARCHAR(50),
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(student_user_id, course_id, session_id)
);

-- 8. ATTENDANCE SESSIONS & RECORDS
CREATE TABLE attendance_sessions (
    session_record_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(course_id),
    faculty_user_id UUID REFERENCES users(user_id),
    lecture_date DATE NOT NULL,
    lecture_date_bs VARCHAR(15),
    dynamic_qr_hash VARCHAR(100),
    qr_expires_at TIMESTAMP WITH TIME ZONE
);

CREATE TABLE attendance_entries (
    entry_id SERIAL PRIMARY KEY,
    session_record_id INT REFERENCES attendance_sessions(session_record_id) ON DELETE CASCADE,
    student_user_id UUID REFERENCES users(user_id),
    status CHAR(1) NOT NULL CHECK (status IN ('P', 'A', 'L', 'E')), -- Present, Absent, Late, Excused
    verified_method VARCHAR(20) DEFAULT 'Manual', -- 'Manual', 'QR_Dynamic'
    recorded_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 9. ASSIGNMENTS & RUBRICS
CREATE TABLE assignments (
    assignment_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(course_id),
    faculty_user_id UUID REFERENCES users(user_id),
    title VARCHAR(255) NOT NULL,
    instructions TEXT,
    max_marks NUMERIC(5,2) DEFAULT 25.00,
    due_date TIMESTAMP WITH TIME ZONE NOT NULL,
    due_date_bs VARCHAR(20),
    rubric_json JSONB,
    late_penalty_percent NUMERIC(4,2) DEFAULT 5.00,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 10. ASSIGNMENT SUBMISSIONS & MARKS
CREATE TABLE submissions (
    submission_id SERIAL PRIMARY KEY,
    assignment_id INT REFERENCES assignments(assignment_id) ON DELETE CASCADE,
    student_user_id UUID REFERENCES users(user_id),
    file_url VARCHAR(500),
    student_notes TEXT,
    submitted_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP,
    is_late BOOLEAN DEFAULT FALSE,
    awarded_marks NUMERIC(5,2),
    faculty_feedback TEXT,
    graded_by UUID REFERENCES users(user_id),
    graded_at TIMESTAMP WITH TIME ZONE
);

-- 11. QUIZZES & QUESTION BANKS
CREATE TABLE quizzes (
    quiz_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(course_id),
    quiz_title VARCHAR(200) NOT NULL,
    duration_minutes INT DEFAULT 15,
    is_published BOOLEAN DEFAULT FALSE
);

CREATE TABLE questions (
    question_id SERIAL PRIMARY KEY,
    quiz_id INT REFERENCES quizzes(quiz_id) ON DELETE CASCADE,
    question_text TEXT NOT NULL,
    options_json JSONB NOT NULL,
    correct_answer VARCHAR(10) NOT NULL,
    explanation TEXT
);

-- 12. LEARNING RESOURCES
CREATE TABLE learning_resources (
    resource_id SERIAL PRIMARY KEY,
    course_id INT REFERENCES courses(course_id),
    uploader_user_id UUID REFERENCES users(user_id),
    title VARCHAR(255) NOT NULL,
    resource_type VARCHAR(50) NOT NULL, -- 'Note', 'Presentation', 'QuestionBank', 'Paper'
    file_path VARCHAR(500) NOT NULL,
    file_size_kb INT,
    downloads_count INT DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 13. RESEARCH PROJECTS
CREATE TABLE research_projects (
    project_id SERIAL PRIMARY KEY,
    title VARCHAR(300) NOT NULL,
    pi_user_id UUID REFERENCES users(user_id),
    co_pi_names TEXT,
    grant_amount_npr NUMERIC(10,2),
    status VARCHAR(50) DEFAULT 'Active', -- 'Proposal', 'Approved', 'Active', 'Completed'
    rec_clearance_status VARCHAR(50) DEFAULT 'Pending',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 14. KAPILVASTU HERITAGE ARCHIVE RECORDS
CREATE TABLE heritage_records (
    record_id SERIAL PRIMARY KEY,
    site_name VARCHAR(200) NOT NULL,
    site_category VARCHAR(50) NOT NULL, -- 'Shakya', 'Monastic', 'Ecology', 'Ethnography'
    historical_period VARCHAR(100),
    latitude NUMERIC(9,6),
    longitude NUMERIC(9,6),
    description TEXT,
    archival_photos JSONB,
    oral_history_audio_url VARCHAR(500),
    is_public BOOLEAN DEFAULT TRUE,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 15. AUDIT LOGS
CREATE TABLE audit_logs (
    log_id BIGSERIAL PRIMARY KEY,
    user_id UUID REFERENCES users(user_id),
    action VARCHAR(100) NOT NULL,
    details TEXT,
    ip_address VARCHAR(45),
    timestamp TIMESTAMP WITH TIME ZONE DEFAULT CURRENT_TIMESTAMP
);

-- 16. INDEXES FOR HIGH-PERFORMANCE QUERYING
CREATE INDEX idx_enrollments_student ON enrollments(student_user_id);
CREATE INDEX idx_attendance_course_date ON attendance_sessions(course_id, lecture_date);
CREATE INDEX idx_submissions_assignment ON submissions(assignment_id);
CREATE INDEX idx_audit_timestamp ON audit_logs(timestamp DESC);

Relational Entity Relationships:

• users (1) ── (M) enrollments ── (1) courses
• courses (1) ── (M) assignments ── (M) submissions ── (1) users (students)
• courses (1) ── (M) attendance_sessions ── (M) attendance_entries
• courses (1) ── (M) learning_resources
• users (faculty) (1) ── (M) research_projects
• heritage_records (Tagged by site, coordinates, consent level)
• audit_logs (Records all critical administrative and grade updates)
